     Stack<Character> st = new Stack<>();
        for(int i = 0; i < n; i++){
            char ch = s.charAt(i);
            if(ch == '(' || ch == '[' || ch == '{'){
                st.push(ch);
            } else {
                if(st.size() == 0){ return false; }
                if(st.size() != 0 && ch == ')' && st.peek() != '(')
                return false;
                if(st.size() != 0 && ch == '}' && st.peek() != '{')
                return false;
                if(st.size() != 0 && ch == ']' && st.peek() != '[')
                return false;
                st.pop();
            }
        }
        return st.size()==0;
    }
