class Solution {
    public int[] maxDepthAfterSplit(String seq) {
        // "(()())"
        // {0,0} {1,0} 
        // ()(())()
        // 
        int d = 0;
        int length = seq.length();
        int[] ans = new int[length];
        for (int i = 0; i < length; i++) {
            if (seq.charAt(i) == '(') {
                ++d;
                ans[i] = d % 2;
            } else {
                // System.out.println(d + " " + i);
                ans[i] = d % 2;
                --d;
            }
        }
        return ans;
    }
}
