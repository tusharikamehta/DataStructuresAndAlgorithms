r> map = new HashMap<>();
        map.put(0, 1);
        // a b c d
        // c-a -> prefix sum of b,c
        // (c-a)%k=0 c%k = a%k
        // 4 9 9 7 4 5
        int ans=0;
        for(int i = 0; i < n; i++){
            int currPrefixSum=prefix[i];
            int rem = currPrefixSum%k;
            if(rem < 0) rem += k;
            if(map.containsKey(rem)){
                ans += map.get(rem);
            }
            map.put(rem, map.getOrDefault(rem, 0)+1);
        }
        // map[4]=1
        // 
        return ans;
    }
