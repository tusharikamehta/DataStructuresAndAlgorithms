class Solution {
    public boolean canTransform(int[] source, int[] target) {
        // delta -> we can choose max val from target?
        int n = source.length;
        int m = target.length;
        if(n != m) return false;
        // delta could be max => sum of array
        // [1,2,3] [0,2,4]
        // i!=j
        // arr[i]+arr[j]-delta = target[i]
        // arr[j]=delta
        // delta = [arr[i], totalSum]
        long sum=0;
        for(int i = 0; i < n; i++){
            sum+=source[i];
            sum-=target[i];
        }
        return sum==0;
    }
}
