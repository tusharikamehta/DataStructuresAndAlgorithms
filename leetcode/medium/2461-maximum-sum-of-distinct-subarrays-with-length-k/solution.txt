            sum += nums[ei];
            while (si <= ei && ei - si + 1 > k) {
                sum -= nums[si];
                map.put(nums[si], map.get(nums[si]) - 1);
                if (map.get(nums[si]) == 0) {
                    map.remove(nums[si]);
                }
                si++;
            }
            if (map.size() == k) {
                maxSum = Math.max(maxSum, sum);
            }
            ei++;
        }
        return maxSum == Long.MIN_VALUE ? 0 : maxSum;
