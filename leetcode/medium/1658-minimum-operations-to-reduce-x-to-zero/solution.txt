       // 20 -> 5,0
        // 3,2,20,1,1,3
        // 2,20,1,1,3 -> 1,7
        // 2,20,1,1 -> 2,4
        // 20,1,1 -> 3,2
        int n = nums.length;
        int si = 0, ei=0;
        int sum=0;
        // longest subarray with the sum = totalSum-x
        for(int i = 0; i < n; i++){
            sum += nums[i];
        }
        int tar=sum-x;
        int maxLen=0;
        int cs=0;
        boolean found=false;
        while(ei < n){
            cs += nums[ei];
            while(si <= ei && cs > tar){
                cs -= nums[si];
                si++;
            }
            if(cs == tar){
                found=true;
                maxLen = Math.max(maxLen, ei-si+1);
            }
            ei++;
        }
        return found ? n-maxLen : -1;
    }
